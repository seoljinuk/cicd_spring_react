          node-version: '18'

      # =====================================================

      - name: Build React Docker Image
        run: |
          echo "===== React Build Start ====="
          docker build \
            -t "${{ secrets.DOCKER_REACT_IMAGE }}" \
            -f react_coffee/DockerfileReact \
            react_coffee
          echo "===== React Build Done ====="

      # =====================================================
      # ?? SPRING SECTION
      # =====================================================

      - name: Build Spring (Maven)
        run: |
          echo "===== Spring Maven Build Start ====="
          cd spring_coffee
          chmod +x mvnw
          ./mvnw clean package -DskipTests
          echo "===== Spring Maven Build Done ====="

      - name: Login to DockerHub
        run: |
          echo "${{ secrets.DOCKER_PASSWORD }}" | \
          docker login -u "${{ secrets.DOCKER_USERNAME }}" --password-stdin

      - name: Build Spring Docker Image
        run: |
          echo "===== Spring Docker Build Start ====="
          cd spring_coffee
          docker build \
            -t "${{ secrets.DOCKER_SPRING_IMAGE }}" \
            -f DockerfileSpring .
          echo "===== Spring Docker Build Done ====="

      # =====================================================
      # ?? PUSH SECTION
      # =====================================================

      - name: Push Spring Image
        run: docker push "${{ secrets.DOCKER_SPRING_IMAGE }}"

      - name: Push React Image

다음 파일은 CI/CD를 위한 cd.yml 파일입니다.

name: CD

on:
  workflow_run:
    workflows: ["CI"]
    types:
      - completed

jobs:
  deploy:
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    runs-on: ubuntu-latest

    steps:
      - name: Deploy to EC2
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ubuntu
          key: ${{ secrets.EC2_KEY }}
          script: |
            cd ~/spring_react
            
            export DOCKER_SPRING_IMAGE=${{ secrets.DOCKER_SPRING_IMAGE }}
            export DOCKER_REACT_IMAGE=${{ secrets.DOCKER_REACT_IMAGE }}            

            echo "===== Pull latest Docker images ====="
            docker compose pull

            echo "===== Restart containers ====="
            docker compose down
            docker compose up -d
