name: CI

on:
  push:
    branches: [ master ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source    
        uses: actions/checkout@v3

      # JDK 21 #ss
      - name: Set up JDK 21
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '21'

      # Node 설치
      - name: Set up Node
        uses: actions/setup-node@v4
        with:
          node-version: '18'

      # React build
      #- name: Build React
      #  run: |
      #    cd react_coffee
      #    npm ci
      #    npm run build

      # Spring build (Maven)
      - name: Build Spring
        run: |
          cd spring_coffee
          chmod +x mvnw
          ./mvnw clean package -DskipTests

      # Docker build 테스트 (선택)
      - name: Docker build test
        run: docker compose build

name: CD

on:
  workflow_run:
    workflows: ["CI"]   # CI 끝난 후 실행
    types:
      - completed

jobs:
  deploy:
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    runs-on: ubuntu-latest

    steps:
      - name: Deploy to EC2
        uses: appleboy/ssh-action@v1
        with:
          host: ${{ secrets.EC2_HOST }}
          username: ubuntu
          key: ${{ secrets.EC2_KEY }}
          script: |
            cd ~/spring_react
            git pull origin master
            docker compose down
            docker compose up -d --build
