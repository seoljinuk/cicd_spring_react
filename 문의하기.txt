name: CI AWS Cloud Spring-React

on:
  push:
    branches: [ master ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout source
        uses: actions/checkout@v4

      # =====================================================
      # ?? COMMON SETUP
      # =====================================================

      - name: Set up JDK 21 (Spring)
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '21'

      - name: Set up Node (React)
        uses: actions/setup-node@v4
        with:
          node-version: '18'

      # =====================================================
      # ?? REACT SECTION 444
      # =====================================================

      - name: Build React Docker Image
        run: |
          echo "===== React Build Start ====="
          docker build \
            -t "${{ secrets.DOCKER_REACT_IMAGE }}" \
            -f react_coffee/DockerfileReact \
            react_coffee
          echo "===== React Build Done ====="

      # =====================================================
      # ?? SPRING SECTION
      # =====================================================

      - name: Build Spring (Maven)
        run: |
          echo "===== Spring Maven Build Start ====="
          cd spring_coffee
          chmod +x mvnw
          ./mvnw clean package -DskipTests
          echo "===== Spring Maven Build Done ====="

      - name: Login to DockerHub
        run: |
          echo "${{ secrets.DOCKER_PASSWORD }}" | \
          docker login -u "${{ secrets.DOCKER_USERNAME }}" --password-stdin

      - name: Build Spring Docker Image
        run: |
          echo "===== Spring Docker Build Start ====="
          cd spring_coffee
          docker build \
            -t "${{ secrets.DOCKER_SPRING_IMAGE }}" \
            -f DockerfileSpring .
          echo "===== Spring Docker Build Done ====="

      # =====================================================
      # ?? PUSH SECTION
      # =====================================================

      - name: Push Spring Image
        run: docker push "${{ secrets.DOCKER_SPRING_IMAGE }}"

      - name: Push React Image
        run: docker push "${{ secrets.DOCKER_REACT_IMAGE }}"

name: CD AWS Cloud Spring-React

on:
  workflow_run:
    workflows: ["CI AWS Cloud Spring-React"]
    types:
      - completed

jobs:

  deploy:

    if: ${{ github.event.workflow_run.conclusion == 'success' }}

    runs-on: ubuntu-latest

    strategy:
      matrix:
        host:
          - ${{ secrets.EC2_HOST_2A }}
          - ${{ secrets.EC2_HOST_2C }}

    steps:

      - name: Deploy to EC2

        uses: appleboy/ssh-action@v1

        with:

          host: ${{ matrix.host }}

          username: ubuntu

          key: ${{ secrets.EC2_KEY }}

          script: |

            cd ~/spring_react

            export DOCKER_SPRING_IMAGE=${{ secrets.DOCKER_SPRING_IMAGE }}

            export DOCKER_REACT_IMAGE=${{ secrets.DOCKER_REACT_IMAGE }}

            docker compose pull

            # docker compose down # down 하면 잠깐 서비스 중단 발생.

            docker compose up -d --remove-orphans
			
			docker image prune -af

    listen 80;

    location / {

        root /usr/share/nginx/html;
        index index.html;
        try_files $uri /index.html;
    }

    location /api/ {

        proxy_pass http://backend:9000/;

        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
